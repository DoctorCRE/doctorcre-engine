CREATE VIEW imported_change AS (

    visible.changed_by_job,
    visible.changed_for_user_id,

     CROSS JOIN LATERAL jsonb_populate_record(c.*, jsonb_build_object('source_event', (c.source_event - ARRAY['human_quote'::text, 'agent_rationale'::text]))) visible(id, table_name, record, erased_fields, old_value_absent, new_value_absent, field, source_event, changed_by_user_id, changed_by_job, changed_for_user_id, changed_at))
