CREATE VIEW deal_imported_history AS (
 SELECT h.deal_id,
    jsonb_agg(jsonb_build_object('id', h.id, 'occurred_at', h.occurred_at, 'recorded_at', h.recorded_at, 'actor', COALESCE(u.slug, h.changed_by_job), 'verb', h.verb, 'field', h.field, 'imported', true, 'old_value', display.old_value, 'new_value', display.new_value) ORDER BY h.recorded_at, h.id) AS events
   FROM ((imported_deal_history h
     LEFT JOIN app_user u ON ((u.id = h.changed_by_user_id)))
     CROSS JOIN LATERAL ( SELECT jsonb_object_agg(label.key, f.value) FILTER (WHERE s.is_before) AS old_value,
            jsonb_object_agg(label.key, f.value) FILTER (WHERE (NOT s.is_before)) AS new_value
           FROM ((( VALUES (true,to_jsonb(h.old_value),h.old_keys), (false,to_jsonb(h.new_value),h.new_keys)) s(is_before, payload, keys)
             CROSS JOIN LATERAL jsonb_each(s.payload) f(key, value))
             CROSS JOIN LATERAL ( SELECT
                        CASE
                            WHEN ((h.verb = 'new-deal'::text) AND (f.key = 'deal_type'::text) AND ((f.value #>> '{}'::text[]) = ANY (ARRAY['relocation'::text, 'startup'::text, 'additional_office'::text, 'expansion'::text, 'renewal'::text]))) THEN 'reason'::text
                            ELSE f.key
                        END AS key) label)
          WHERE ((f.key = ANY (s.keys)) AND (f.key <> ALL (ARRAY['client_portal_url'::text, 'notes_path'::text])) AND ((h.verb <> 'set-next-action'::text) OR (f.key <> 'summary'::text)) AND (NOT ((f.key = 'source'::text) AND ((f.value #>> '{}'::text[]) ~* '^\s*https?://\S+\s*$'::text))) AND ((f.value <> ALL (ARRAY['null'::jsonb, '"None"'::jsonb, '""'::jsonb, '{"pdf": null, "working": null}'::jsonb])) OR ((NOT s.is_before) AND (f.value = 'null'::jsonb) AND (f.key = ANY (h.old_keys)) AND ((to_jsonb(h.old_value) -> f.key) <> ALL (ARRAY['null'::jsonb, '"None"'::jsonb, '""'::jsonb, '{"pdf": null, "working": null}'::jsonb])))))) display)
  WHERE (NOT (EXISTS ( SELECT
           FROM (unnest(ARRAY[(h.new_value).attachments.pdf, (h.new_value).attachments.working]) attachment(id)
             LEFT JOIN document doc ON (((doc.id = attachment.id) AND (doc.deal_id = h.deal_id))))
          WHERE ((h.verb = 'backfill-document-attachments'::text) AND (attachment.id IS NOT NULL))
         HAVING ((count(*) > 0) AND (count(*) = count(doc.id))))))
  GROUP BY h.deal_id
)
