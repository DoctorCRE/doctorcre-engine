```sql
CREATE VIEW home_update AS (
 SELECT visible.id,
    visible.user_id,
    visible.deal_id,
    visible.kind,
    visible.what,
    visible.checklist_item_ref,
    visible.due_on,
    visible.property_id,
    visible.created_at,
    visible.held_until,
    visible.read_at,
    visible.changed_by_user_id,
    visible.changed_by_job,
    visible.changed_for_user_id,
    visible.changed_at
   FROM ((home_update u
     LEFT JOIN next_step s ON (((s.deal_id = u.deal_id) AND (('deal-item:'::text || s.id) = u.checklist_item_ref))))
     CROSS JOIN LATERAL jsonb_populate_record(u.*, jsonb_build_object('what',
        CASE
            WHEN ((u.checklist_item_ref ~~ 'deal-item:%'::text) AND (s.id IS NULL)) THEN 'A deal checklist item is due. Open the deal checklist for details.'::text
            ELSE u.what
        END)) visible(id, user_id, deal_id, kind, what, checklist_item_ref, due_on, property_id, created_at, held_until, read_at, changed_by_user_id, changed_by_job, changed_for_user_id, changed_at))
  WHERE (u.user_id = (NULLIF(current_setting('engine.user_id'::text, true), ''::text))::uuid)
)
```
