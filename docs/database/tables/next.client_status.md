<summary><strong>Table Definition</strong></summary>

```sql
CREATE VIEW client_status AS (
 SELECT party_id,
        CASE
            WHEN (EXISTS ( SELECT
               FROM deal_party dp
              WHERE ((dp.party_id = c.party_id) AND (dp.role = 'client'::text) AND (COALESCE(( SELECT p.phase
                       FROM deal_phase_change p
                      WHERE (p.deal_id = dp.deal_id)
                      ORDER BY p.happened_at DESC
                     LIMIT 1), ''::text) <> ALL (ARRAY['closed'::text, 'dead'::text]))))) THEN 'active_deal'::text
            WHEN (EXISTS ( SELECT
               FROM (deal_party dp
                 JOIN deal_phase_change p USING (deal_id))
              WHERE ((dp.party_id = c.party_id) AND (dp.role = 'client'::text) AND (p.phase = 'closed'::text)))) THEN 'past_client'::text
            ELSE status
        END AS status
   FROM client c
)
```
