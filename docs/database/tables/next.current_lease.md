```sql
CREATE VIEW current_lease AS (
 SELECT v.id,
    v.created_at,
    v.lease_id,
    v.version,
    v.previous_version,
    v.offer_id,
    v.document_id,
    v.executed_on,
    v.commencement_on,
    v.expiration_on,
    v.rent_per_sf_yr,
    v.nnn_per_sf_yr,
    v.term_months,
    v.ti_per_sf,
    v.free_rent_months,
    v.escalation_pct,
    v.rent_starts_on,
    v.rent_days_after_delivery,
    v.expense_structure,
    v.other_terms,
    l.deal_id,
    l.party_id,
    l.property_id
   FROM (lease_version v
     JOIN lease l ON ((l.id = v.lease_id)))
  WHERE (NOT (EXISTS ( SELECT
           FROM lease_version newer
          WHERE ((newer.lease_id = v.lease_id) AND (newer.version > v.version)))))
)
```
