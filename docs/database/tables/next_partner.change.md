```sql
CREATE VIEW change AS (
 SELECT id,
    table_name,
    record,
    action,
        CASE
            WHEN (table_name = 'home_update'::text) THEN (old_value - 'what'::text)
            ELSE old_value
        END AS old_value,
        CASE
            WHEN (table_name = 'home_update'::text) THEN (new_value - 'what'::text)
            ELSE new_value
        END AS new_value,
    merge_of,
    changed_by_user_id,
    changed_by_job,
    changed_for_user_id,
    changed_at
   FROM change c
  WHERE
        CASE
            WHEN (table_name = 'lead_stage_move'::text) THEN (EXISTS ( SELECT
               FROM (lead_stage_move m
                 LEFT JOIN next_partner.activity a ON ((a.id = m.activity_id)))
              WHERE ((m.id = ((c.record ->> 'id'::text))::uuid) AND ((m.activity_id IS NULL) OR (a.id IS NOT NULL)))))
            WHEN (table_name = 'home_update'::text) THEN (EXISTS ( SELECT
               FROM next_partner.home_update u
              WHERE (u.id = ((c.record ->> 'id'::text))::uuid)))
            ELSE true
        END
)
```
