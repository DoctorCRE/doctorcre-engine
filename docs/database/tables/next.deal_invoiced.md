```sql
CREATE VIEW deal_invoiced AS (
 SELECT deal_id,
    min(invoiced_on) AS invoiced_on
   FROM ( SELECT commission_payment.deal_id,
            commission_payment.invoiced_on
           FROM commission_payment
          WHERE (commission_payment.invoiced_on IS NOT NULL)
        UNION ALL
         SELECT imported_deal_invoiced.deal_id,
            imported_deal_invoiced.invoiced_on
           FROM imported_deal_invoiced) dated
  GROUP BY deal_id
)
```
