# next.lease_ending

## Description

<summary><strong>Table Definition</strong></summary>

```sql
CREATE VIEW lease_ending AS (
 SELECT id,
    created_at,
    lease_id,
    version,
    previous_version,
    offer_id,
    document_id,
    executed_on,
    commencement_on,
    expiration_on,
    rent_per_sf_yr,
    nnn_per_sf_yr,
    term_months,
    ti_per_sf,
    free_rent_months,
    escalation_pct,
    rent_starts_on,
    rent_days_after_delivery,
    expense_structure,
    other_terms,
    deal_id,
    party_id,
    property_id
   FROM current_lease c
  WHERE (NOT (EXISTS ( SELECT
           FROM ((deal d
             JOIN lease l ON ((l.deal_id = d.id)))
             JOIN lease_version v ON ((v.lease_id = l.id)))
          WHERE (d.renews_lease_id = c.lease_id))))
)
```
