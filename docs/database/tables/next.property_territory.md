CREATE VIEW property_territory AS (
 SELECT p.id AS property_id,
    piece.territory_id
   FROM (property p
     JOIN territory_piece piece ON (((piece.state_fips)::text = "left"((p.county_fips)::text, 2))))
UNION
 SELECT p.id AS property_id,
    piece.territory_id
   FROM ((property p
     JOIN market_county m ON (((m.county_fips)::text = (p.county_fips)::text)))
     JOIN territory_piece piece ON ((piece.market_id = m.market_id)))
UNION
 SELECT p.id AS property_id,
    piece.territory_id
   FROM (property p
     JOIN territory_piece piece ON ((miles_between((p.latitude)::double precision, (p.longitude)::double precision, (piece.base_latitude)::double precision, (piece.base_longitude)::double precision) <= (piece.miles)::double precision)))
)
