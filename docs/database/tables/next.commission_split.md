CREATE VIEW commission_split AS (
 WITH closed AS (
         SELECT d.id AS deal_id,
            row_number() OVER (PARTITION BY d.owner_user_id ORDER BY last.happened_at, d.id) AS number
           FROM (deal d
             CROSS JOIN LATERAL ( SELECT deal_phase_change.phase,
                    deal_phase_change.happened_at
                   FROM deal_phase_change
                  WHERE (deal_phase_change.deal_id = d.id)
                  ORDER BY deal_phase_change.happened_at DESC
                 LIMIT 1) last)
          WHERE (last.phase = 'closed'::text)
        ), users AS (
         SELECT m_1.deal_id,
            d.owner_user_id,
            ladder.partner_user_id,
            COALESCE(m_1.closing_share, ( SELECT t.closing_share
                   FROM commission_tier t
                  WHERE ((t.user_id = d.owner_user_id) AND (t.from_deal <= closed.number))
                  ORDER BY t.from_deal DESC
                 LIMIT 1)) AS closing_share,
            ((1)::numeric - COALESCE(( SELECT sum(commission_share.share) AS sum
                   FROM commission_share
                  WHERE (commission_share.deal_id = m_1.deal_id)), (0)::numeric)) AS left_share
           FROM (((commission m_1
             JOIN deal d ON ((d.id = m_1.deal_id)))
             JOIN LATERAL ( SELECT commission_tier.partner_user_id
                   FROM commission_tier
                  WHERE (commission_tier.user_id = d.owner_user_id)
                 LIMIT 1) ladder ON (true))
             LEFT JOIN closed USING (deal_id))
        ), parts AS (
         SELECT commission_share.deal_id,
            commission_share.party_id,
            commission_share.kind,
            commission_share.share,
            0 AS part,
            commission_share.id
           FROM commission_share
        UNION ALL
         SELECT users.deal_id,
            users.owner_user_id,
            'closing_user'::text,
            (users.left_share * users.closing_share),
            1,
            NULL::uuid
           FROM users
        UNION ALL
         SELECT users.deal_id,
            users.partner_user_id,
            'partner'::text,
            (users.left_share * ((1)::numeric - users.closing_share)),
            2,
            NULL::uuid
           FROM users
        )
 SELECT p.deal_id,
    p.party_id,
    p.kind,
    p.share,
    (round((m.gross * sum(p.share) OVER running), 2) - round((m.gross * (sum(p.share) OVER running - p.share)), 2)) AS amount
   FROM (parts p
     JOIN commission m USING (deal_id))
  WINDOW running AS (PARTITION BY p.deal_id ORDER BY p.part, p.id)
)
