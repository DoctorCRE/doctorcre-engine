# next.vendor_track_record

```sql
CREATE VIEW vendor_track_record AS (
 WITH worked AS (
         SELECT dp.party_id,
            dp.deal_id,
            ( SELECT c.phase
                   FROM deal_phase_change c
                  WHERE (c.deal_id = dp.deal_id)
                  ORDER BY c.happened_at DESC
                 LIMIT 1) AS phase
           FROM (deal_party dp
             JOIN role r ON ((r.slug = dp.role)))
          WHERE (('vendor'::text = r.slug) OR ('vendor'::text = r.category_of))
        )
 SELECT v.party_id,
    count(DISTINCT w.deal_id) AS deals_worked,
    count(DISTINCT w.deal_id) FILTER (WHERE (w.phase = 'closed'::text)) AS deals_closed,
    count(DISTINCT w.deal_id) FILTER (WHERE (w.phase = 'dead'::text)) AS deals_dead,
    ( SELECT count(DISTINCT introduction.deal_id) AS count
           FROM introduction
          WHERE ((introduction.vendor_id = v.party_id) AND (introduction.made_by = 'vendor'::text))) AS deals_it_introduced,
    ( SELECT count(DISTINCT introduction.deal_id) AS count
           FROM introduction
          WHERE ((introduction.vendor_id = v.party_id) AND (introduction.made_by = 'team'::text))) AS deals_we_introduced
   FROM (party_role v
     LEFT JOIN worked w USING (party_id))
  WHERE (v.role = 'vendor'::text)
  GROUP BY v.party_id
)
```
